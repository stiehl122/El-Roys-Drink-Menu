## Task 1: Replace `confirm()` on Item Removal with Undo Toast (#151)

**Files:**
- Modify: `app.js` — `removeItem()`, `showToast()`, and the `document.addEventListener` Escape key handler
- Modify: `style.css` — add `.toast.undo` variant and `.toast-undo-btn` button style

**Context:** `removeItem(catId, itemId)` currently calls `confirm()`, which blocks the UI thread. The fix is to remove the item immediately, show a toast with an Undo button, and restore the item if clicked within ~5 seconds. The existing `showToast` function sets `t.textContent` directly — it needs extending to optionally accept an action callback. The item wrapper DOM ID is `wrapper-${item.id}`. `renderManagerItems(catId)` re-renders the item list. `updateDraftIndicator()` updates the send button badge. Both must be called after undo.

- [ ] **Step 1: Extend `showToast` to support an optional action button**

In `app.js`, replace the `showToast` function with:

```javascript
// ─── TOAST ───────────────────────────────────────────────────────────────────
let _toastUndoTimer = null;
function showToast(msg, type='info', undoCallback=null) {
  if (_toastUndoTimer) { clearTimeout(_toastUndoTimer); _toastUndoTimer = null; }
  const t = document.getElementById('toast');
  t.className = `toast ${type} show`;
  if (undoCallback) {
    t.innerHTML = `<span>${escHtml(msg)}</span><button class="toast-undo-btn" onclick="_toastUndo()">Undo</button>`;
    window._toastUndoCallback = undoCallback;
    _toastUndoTimer = setTimeout(() => { t.classList.remove('show'); window._toastUndoCallback = null; }, 5000);
  } else {
    t.textContent = msg;
    window._toastUndoCallback = null;
    _toastUndoTimer = setTimeout(() => t.classList.remove('show'), 3200);
  }
}
function _toastUndo() {
  if (typeof window._toastUndoCallback === 'function') {
    window._toastUndoCallback();
    window._toastUndoCallback = null;
  }
  const t = document.getElementById('toast');
  t.classList.remove('show');
  if (_toastUndoTimer) { clearTimeout(_toastUndoTimer); _toastUndoTimer = null; }
}
```

Existing callers pass only `msg` and `type` — the third parameter defaults to `null`, so all existing `showToast(...)` calls continue to work unchanged. Non-undo toasts keep the original 3200ms timeout.

- [ ] **Step 2: Rewrite `removeItem` to remove immediately and offer undo**

Replace the `removeItem` function with:

```javascript
function removeItem(catId, itemId) {
  const item = findItem(catId, itemId);
  if (!item) return false;
  item.onMenu = false;
  invalidateDiff();
  renderManagerItems(catId);
  updateDraftIndicator();
  const removedName = item.name;
  showToast(`"${removedName}" removed`, 'info', () => {
    item.onMenu = true;
    invalidateDiff();
    renderManagerItems(catId);
    updateDraftIndicator();
    showToast(`"${removedName}" restored`, 'success');
  });
  return true;
}
```

```javascript
  if (!isManagerMode) return;

    e.preventDefault();
    saveMenu();
  }
});
```

## Task 3: Relative "Updated X ago" Timestamp (#153)

**Files:**
- Modify: `app.js` — add `formatRelativeTime()`, update `updateLastUpdatedLabel()` and `renderFooter()`, add a live-refresh interval

**Context:** `getLastUpdatedTs()` returns a Unix timestamp in milliseconds or `null`. `updateLastUpdatedLabel()` sets `#last-updated-label` in the header. `renderFooter()` sets `#footer-last-updated`. Both currently call `formatUpdatedAt(ts, prefix)` which returns an absolute date string. The goal: show relative time as displayed text, absolute date as a `title` tooltip. A `setInterval` refreshes the label every 60 seconds.

- [ ] **Step 1: Add `formatRelativeTime()` helper**

In `app.js`, immediately after `formatUpdatedAt`, add:

```javascript
function formatRelativeTime(ts) {
  if (!ts) return null;
  const diffMs = Date.now() - parseInt(ts);
  const diffMin = Math.floor(diffMs / 60000);
  if (diffMin < 1)  return 'just now';
  if (diffMin < 60) return `${diffMin} minute${diffMin === 1 ? '' : 's'} ago`;
  const diffHr = Math.floor(diffMin / 60);
  if (diffHr < 24)  return `${diffHr} hour${diffHr === 1 ? '' : 's'} ago`;
  const diffDay = Math.floor(diffHr / 24);
  if (diffDay < 7)  return `${diffDay} day${diffDay === 1 ? '' : 's'} ago`;
  return formatUpdatedAt(ts, ''); // fall back to absolute for older timestamps
}
```

- [ ] **Step 2: Update `updateLastUpdatedLabel()`**

Replace the function with:

```javascript
function updateLastUpdatedLabel() {
  const ts = getLastUpdatedTs();
  const el = document.getElementById('last-updated-label');
  if (ts) {
    const rel = formatRelativeTime(ts);
    const abs = formatUpdatedAt(ts, 'Last Updated: ');
    el.textContent = `Updated ${rel}`;
    el.title = abs;
  } else {
    el.textContent = 'Last Updated: —';
    el.title = '';
  }
  renderFooter();
}
```

- [ ] **Step 3: Update `renderFooter()`**

Replace the function with:

```javascript
function renderFooter() {
  const vEl = document.getElementById('footer-version');
```

- [ ] **Step 4: Add a live-refresh interval**

## Task 4: Shake Animation + Color Flash on 86 Toggle (#154)

**Files:**
- Modify: `style.css` — add `@keyframes item-shake`, CSS variables, and animation classes
- Modify: `app.js` — update `toggle86()` to trigger the animation

**Context:** `toggle86(catId, itemId)` mutates `item.eightySixed` then calls `renderManagerItems(catId)` which rebuilds the entire item list. Because the DOM is rebuilt, the animation class must be added to the newly-rendered element after `renderManagerItems` returns. The item wrapper DOM ID is `wrapper-${item.id}`. The animation should shake the row and flash the background — red-tinted when 86'ing, green-tinted when restoring. Remove the class after ~400ms via `setTimeout`.

- [ ] **Step 2: Add keyframe and animation classes**

- [ ] **Step 3: Update `toggle86()` to trigger the animation**

Replace the `toggle86` function with:

```javascript
function toggle86(catId, itemId) {
  const item = findItem(catId, itemId);
  if (!item) return;
  item.eightySixed = !item.eightySixed;
  invalidateDiff();
  renderManagerItems(catId);
  updateDraftIndicator();
  // Trigger animation on the newly-rendered wrapper
  const wrapper = document.getElementById('wrapper-' + itemId);
  if (wrapper) {
    const cls = item.eightySixed ? 'flash-86' : 'flash-restore';
    wrapper.classList.add(cls);
    setTimeout(() => wrapper.classList.remove(cls), 400);
  }
  showToast(item.eightySixed ? "🚫 Marked 86'd — send update to notify group" : `↩ Marked ${restoreLabel(catId)} — send update to notify group`, 'info');
}
```

## Concerns and Ambiguities

- **#151 — Undo timeout:** 5 seconds chosen. Adjust the `5000` in `_toastUndoTimer` if a different window is preferred.
- **#151 — `renameItem` blank-name delete:** This also calls `removeItem`, so will now show the undo toast instead of `confirm()`. This is correct behavior.
- **#153 — Live refresh scope:** The `setInterval` fires unconditionally (including manager mode). Harmless — `updateLastUpdatedLabel` is safe to call at any time.
- **#154 — CSS convention:** Raw `rgba()` literals avoided by declaring `--flash-86-bg` / `--flash-restore-bg` variables. Implementor must add these to both `:root` and `body.manager-mode`.
- **#155 — Placeholder length:** If a category's `placeholder` is long and wraps awkwardly, the implementor may omit it and keep "Nothing here yet." — the visible input above implies the action.
